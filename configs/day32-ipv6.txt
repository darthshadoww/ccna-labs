! Day 32 — IPv6 Part 2 (EUI-64, ipv6 enable, FE80 statics)

! ----- R1 -----
ipv6 unicast-routing

interface GigabitEthernet0/1
 ipv6 address 2001:db8::/64 eui-64
 no shutdown
! → 2001:DB8::230:F2FF:FE36:4502
! → FE80::230:F2FF:FE36:4502

interface GigabitEthernet0/0
 ipv6 enable
 no shutdown
! → FE80 only

ipv6 route 2001:db8:0:1::/64 FE80::201:63FF:FEB0:B802 GigabitEthernet0/0

! ----- R2 -----
ipv6 unicast-routing

interface GigabitEthernet0/1
 ipv6 address 2001:db8:0:1::/64 eui-64
 no shutdown
! → 2001:DB8:0:1:201:63FF:FEB0:B802
! → FE80::201:63FF:FEB0:B802

interface GigabitEthernet0/0
 ipv6 enable
 no shutdown
! → FE80::201:63FF:FEB0:B801

ipv6 route 2001:db8::/64 FE80::230:F2FF:FE36:4502 GigabitEthernet0/0

! ----- PCs -----
! PC1: 2001:DB8::2/64  gw 2001:DB8::230:F2FF:FE36:4502
! PC2: 2001:DB8:0:1::2/64  gw 2001:DB8:0:1:201:63FF:FEB0:B802

! Verify
show ipv6 interface brief
show ipv6 route
! PC1: ping 2001:db8:0:1::2
